#include <omp.h>

#include <stdio.h>
#include <omp.h>

#define N 100000000   // 1億

int main() {
    double start, end;
    long long sum1 = 0, sum2 = 0;

    printf("スレッド数 = %d\n", omp_get_max_threads());

    // -----------------------------
    // ① parallel（自分で分割）
    // -----------------------------
    start = omp_get_wtime();

    #pragma omp parallel
    {
        int tid = omp_get_thread_num();
        int nthreads = omp_get_num_threads();

        long long local = 0;

        for (int i = tid; i < N; i += nthreads) {
            local += i;
        }

        #pragma omp atomic
        sum1 += local;
    }

    end = omp_get_wtime();
    printf("parallel     : sum=%lld  time=%f sec\n", sum1, end - start);


    // -----------------------------
    // ② parallel for（自動分割）
    // -----------------------------
    start = omp_get_wtime();

    #pragma omp parallel for reduction(+:sum2)
    for (int i = 0; i < N; i++) {
        sum2 += i;
    }

    end = omp_get_wtime();
    printf("parallel for : sum=%lld  time=%f sec\n", sum2, end - start);

    return 0;
}
